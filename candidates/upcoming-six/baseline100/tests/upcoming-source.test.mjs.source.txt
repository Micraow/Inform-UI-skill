import test from 'node:test';import assert from 'node:assert/strict';import {readFile,readdir} from 'node:fs/promises';import {createHash} from 'node:crypto';import path from 'node:path';
const root=path.resolve(import.meta.dirname,'..'),at=p=>path.join(root,'candidates/upcoming-three',p),json=async p=>JSON.parse(await readFile(p,'utf8')),sha=b=>createHash('sha256').update(b).digest('hex');
test('upcoming source package is independently pinned and never upgrades accepted counts',async()=>{const p=await json(at('source-package.json')),current=await json(path.join(root,'library-contract.json'));assert.equal(p.libraryRevision,'3858d0a502956d283f3d60ab8c2481f691e50522');assert.equal(current.revision,'26ec211fa529516af3b1523248f5912c45ec64c6');assert.equal(p.acceptedCanonical,53);assert.equal(p.existingPendingCanonical,37);assert.equal(p.protocolNodes,100);assert.deepEqual(p.upcomingCanonical,['create-interactive-poll','email-preview','file-nav-list']);assert.equal(p.verification.acceptancePromotion,false);assert.equal(Object.keys(p.productionAndBuildHashes).length,360);});
test('all generated full/subset/example bytes are locked with unique ownership',async()=>{const p=await json(at('source-package.json'));for(const [file,digest]of Object.entries(p.generatedFiles))assert.equal(sha(await readFile(at(file))),digest,file);const index=await json(at('schema/fragments/index.json'));assert.equal(index.groups.length,11);assert.equal(Object.keys(index.nodeOwners).length,100);for(const type of p.upcomingCanonical)assert.equal(index.nodeOwners[type],'base');assert.equal((await readdir(at('schema/fragments/nodes'))).length,11);});
test('new Base guidance is inline and frozen37 guidance prefix remains exact',async()=>{const baseline=await readFile(at('baseline37/SKILL.md.source.txt'),'utf8'),skill=await readFile(path.join(root,'SKILL.md'),'utf8'),guidance=await readFile(at('guidance-source.md'),'utf8'),marker='<!-- upcoming-three-source-authoring -->';assert.equal(skill.split(marker).length,2);assert.equal(skill.split(marker)[0],baseline+'\n');assert.ok(skill.endsWith(guidance));assert.equal([...guidance.matchAll(/^```json upcoming-only$/gm)].length,2);const lock=await json(path.join(root,'candidates/pending-batch/library-candidate-lock37.json'));for(const file of ['SKILL.md','tests/pending-batch.test.mjs'])assert.equal(sha(await readFile(at('baseline37/'+file+(file==='SKILL.md'?'.source.txt':'')))),lock.candidateFiles[file]);});
test('negative examples distinguish semantic defects from legal incomplete poll drafts',async()=>{const cases=await json(at('rejection-examples.json'));assert.equal(cases.length,23);assert.equal(cases.filter(c=>c.code==='SCHEMA').length,9);for(const code of ['DUPLICATE_ID','READER_TIME','UNSAFE_URL','FILE_PARENT','FILE_CYCLE','FILE_FOLDER','FILE_DEPTH'])assert.ok(cases.some(c=>c.code===code&&c.path.startsWith('/body/0/')));assert.ok(!cases.some(c=>c.name.includes('blank')));});
